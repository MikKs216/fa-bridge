Hlavní funkce skriptu

procedure Init;

//při vytvoření nového formuláře se inicializují všechny údaje (většinou hodnotami z nastavení klienta)

//při vstupu do prázdného údaje, nebo při stisku mezerníku u vyplněného údaje, se nastaví základní hodnota tohoto jednoho údaje

//ini (ssInit) - vyvolá se při vytvoření nového formuláře, při vstupu do nevyplněného údaje a při stisku mezerníku na údaji

//ino - vyvolá se pouze při vytvoření nového formuláře

//def (ssDefault) - vyvolá se pouze při vstupu do nevyplněného údaje nebo při stisku mezerníku na údaji

ini('', value);

ino('', value);

def('', value);

//if GScriptState=ssInit then ...

//if GScriptState=ssDefault then ...

 

 

procedure Calculate;

//výpočty všech údajů

//vyvolá se při opuštění údaje, pokud byl údaj změněn

cal('', value);

 

 

procedure Enable;

//povolování/zakazování editace údajů na základě nějaké podmínky

//pokud je podmínka (condition) splněna, údaj je povolen (enabled)

//vyvolá se při změně každého údaje ve formuláři

ena('', condition);

 

 

function Groups: string;

//definuje seznam skupin checkboxů

//DisableUncheck zajistí, aby byl vždy alespoň jeden checkbox zatržen

//v rámci skupiny jsou údaje odděleny čárkou, skupiny jsou odděleny středníkem

//vyvolá se při změně kteréhokoliv z uvedených údajů

//např.:

Result:='checkboxA1,checkboxA2,checkboxA3,DisableUncheck;checkboxB1,checkboxB2';

 

 

procedure Check;

//kontroly

//pokud je podmínka (condition) splněna, objeví se hlášení (message) v kontrolách

//typHlaseni: 0-chyba(default),1-varování,2-tip,4-chyba EPO,5-varování EPO

chk('', condition, message, typHlaseni);

 

 

procedure OnEndShowChecking;

//vyvolá se, pokud v okně kontrol vybereme řádek

function SysComment: string;

//vrací v řetězci obsah systémové poznámky,

//např.:

Result:='radne';

 

 

function Period: string;

//vrací v řetězci období formuláře

//funkce musí být v každém formuláři

//pokud na formuláři není žádný datumový údaj:

Result:=CalcPeriod('');

//pokud je zdaňovacím obdobím rok:

Result:=CalcPeriod([:Rok:]);

//pokud je zdaňovacím obdobím měsíc a rok:

Result:=CalcPeriod([:Rok:]+','+[:Mesic:]);

//pokud je zdaňovacím obdobím čtvrtletí a rok:

Result:=CalcPeriod([:Rok:]+','+[:Ctvrtleti:]+',Q');

//pokud je zdaňovacím obdobím datum od - do:

Result:=CalcPeriod([:DatumOd:]+','+[:DatumDo:]);

//pokud je na formuláři jen jedno nějaké datum:

Result:=CalcPeriod([:Datum:]);

 

 

function Characteristics: string;

//vrací v řetězci charakteristiky formuláře.

//jednotlivé charakteristiky jsou odděleny novým řádkem (CRLF)

//např.:

Result:='Název firmy';

Result:=Result + 'Adresa firmy';
