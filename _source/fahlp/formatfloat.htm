FormatFloat

Naformátuje desetinného čísla na řetězce.

 

Syntaxe:

function FormatFloat(const Format: string; Value: Extended): string;

 

Parametry:

Format - požadovaný formát

DateTime - číselná hodnota

 

Result:

Naformátuje číslo na požadovaný formát.

Pokud je parametr Format prázdný, vrací původní číslo převedené na řetězec.

 

Unit

Delphi | SysUtils

 

Příklad:

Ukázky použití ve skriptu FA:

 

Format        12345.6789                0.6789        0    

nezadáno        12345.6789                0.6789        0

0                12346                        1                0

#                12346                        1                prázdný řetězec

,0                12 346                1                0

,#                12 346                1                prázdný řetězec

,0.0                12 345.7                0.7                0.0

,#.0                12 345.7                .7                .0

,0.0000        12 345.6789                0.6789        0.0000

,0.00##        12 345.6789                0.6789        0.00

 

var

 float : extended;

 

begin

 // Nastavení našeho desetinného čísla

 float := 1234.567;

 

 

 ShowMessage('##### : '+FormatFloat('#####', float));

 ShowMessage('00000 : '+FormatFloat('00000', float));

 ShowMessage('0     : '+FormatFloat('0'    , float));

 ShowMessage('#,##0 : '+FormatFloat('#,##0', float));

 ShowMessage(',0    : '+FormatFloat(',0'   , float));

 ShowMessage('');

 

 // Zahrnuje desítkovou hodnotu

 ShowMessage('0.#### : '+FormatFloat('0.####', float));

 ShowMessage('0.0000 : '+FormatFloat('0.0000', float));

 ShowMessage('');

 

 // Vědecký formát

 ShowMessage('0.0000000E+00 : '+FormatFloat('0.0000000E+00', float));

 ShowMessage('0.0000000E-00 : '+FormatFloat('0.0000000E-00', float));

 ShowMessage('#.#######E-## : '+FormatFloat('#.#######E-##', float));

 ShowMessage('');

 

 // Zahrnutí textu

 ShowMessage('"Value = "0.0 : '+FormatFloat('"Value = "0.0', float));

 ShowMessage('');

 

 // Různé formátování pro záporná čísla

 ShowMessage('0.0 : '+FormatFloat('0.0'              , -1234.567));

 ShowMessage('0.0 "CR";0.0 "DB" : '+

             FormatFloat('0.0 "CR";0.0 "DB"', -1234.567));

 ShowMessage('0.0 "CR";0.0 "DB" : '+

             FormatFloat('0.0 "CR";0.0 "DB"',  1234.567));

 ShowMessage('');

 

 // Různý formát pro nulovou hodnotu

 ShowMessage('0.0 : '+FormatFloat('0.0' , 0.0));

 ShowMessage('0.0;-0.0;"Nothing" : '+

             FormatFloat('0.0;-0.0;"Nothing"', 0.0));

end;
