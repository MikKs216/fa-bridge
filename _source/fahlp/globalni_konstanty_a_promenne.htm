Globální konstanty a proměnné

Seznam globálních konstant a proměnných, které lze ve skriptu použít:

 

const

 

cCRLF = #13#10;

cCRLFAlternate = #255'#';

cFADecimalSeparator = '.';

cMaxInt = 2147483647;

 

var //proměnné jsou globální pro všechny otevřené formuláře a jde do nich přímo zapisovat

 

Gsl: TStringList; //globalni string list obecne pouzitelny ve scriptech

GInternalDataForPrint: string; //data ktera pripoji po beforeprint, pouziti napr pro zkrazene vykazy

GEPOFilePath: string; //plní se při importu formuláře z nastavení klienta, obsahuje cestu k importovanému souboru

GLoadingBreak: boolean; // nastavi se pokud pro prubehu zobrazeni dialogu Loading je stisknuto Esc

FSelectedDir: string; //naposled vybrany adresar funkci SelectDir
