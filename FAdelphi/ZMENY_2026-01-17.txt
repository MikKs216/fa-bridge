===============================================================================

1. V Delphi IDE:
   Project → Add to Project... → vybrat:
   - EPMF\mfcr_certificate_api.pas
   - EPMF\mfcr_certificate.pas
   - EPMF\mfcr_consts.pas
   - EPMF\mfcr_utils.pas

2. V kódu:
   uses FCommLifecycle, EPMF.mfcr_certificate_api;
   
   var FCommManager: TFCommLifecycle;
   begin
     FCommManager := TFCommLifecycle.Create;
     try
       // Pro VM testování:

       FCommManager.Start;
       // ... práce s certifikáty ...
     finally
       FCommManager.Free;
     end;
   end;
